def utc_timestamp() -> str:
    return datetime.now(timezone.utc).strftime(
        "%Y-%m-%dT%H:%M:%SZ"
    )


def fetch_coingecko_btc_price() -> float:
    """
    Fetch the current Bitcoin price in USD from CoinGecko.

    This uses the keyless public CoinGecko API.
    No API key is required.
    """

    response = requests.get(
        COINGECKO_PRICE_URL,
        params={
            "ids": "bitcoin",
            "vs_currencies": "usd",
        },
        headers={
            "Accept": "application/json",
            "User-Agent": (
                "operational-governance-engine-daemon/1.0"
            ),
        },
        timeout=10,
    )

    response.raise_for_status()

    data = response.json()

    if not isinstance(data, dict):
        raise ValueError(
            f"Unexpected CoinGecko response: {data}"
        )

    bitcoin_data = data.get("bitcoin")

    if not isinstance(bitcoin_data, dict):
        raise ValueError(
            f"Missing bitcoin data in response: {data}"
        )

    if "usd" not in bitcoin_data:
        raise ValueError(
            f"Missing USD price in response: {data}"
        )

    price = float(bitcoin_data["usd"])

    if price <= 0:
        raise ValueError(
            f"Invalid CoinGecko Bitcoin price: {price}"
        )

    return price


def append_daemon_record(record: dict) -> None:
    with open(
        AUDIT_PATH,
        "a",
        encoding="utf-8",
    ) as file:
        file.write(
            json.dumps(
                record,
                sort_keys=True,
            )
            + "\n"
        )
